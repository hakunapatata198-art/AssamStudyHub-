export function getClassBySlug(classSlug) {
  return classes.find((item) => item.slug === classSlug);
}

export function getSubject(classSlug, subjectSlug) {
  return getClassBySlug(classSlug)?.subjects.find((item) => item.slug === subjectSlug);
}

export function getChapter(classSlug, subjectSlug, chapterSlug) {
  return getSubject(classSlug, subjectSlug)?.chapters.find((item) => item.slug === chapterSlug);
}

export function getFeaturedChapters() {
  return classes.flatMap((course) =>
    course.subjects.flatMap((subject) =>
      subject.chapters.slice(0, 1).map((chapter) => ({
        classSlug: course.slug,
        className: course.name,
        subjectSlug: subject.slug,
        subjectName: subject.name,
        ...chapter,
      }))
    )
  );
}
