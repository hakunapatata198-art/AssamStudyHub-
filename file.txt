        function switchClass(classNum) {
            SelectedClass = classNum;
            
            // Manage Sidebar Button Highlighting states
            const buttons = document.querySelectorAll('.class-btn');
            buttons.forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');

            // Update main panel headers
            document.getElementById('currentClassTitle').innerText = `Class ${classNum} Solutions`;
            closeNotesWindow();
        }

        function openSubject(subjectName) {
            document.getElementById('hubView').classList.remove('active-view');
            document.getElementById('notesWindow').classList.add('active-window');
            
            // Set Headers based on context selection
            document.getElementById('notesSubjectHeader').innerText = `Class ${SelectedClass} : ${subjectName}`;
            
            // Provide dynamic content parsing or a fallback placeholder if data lacks entries
            const renderTarget = document.getElementById('notesRenderSpace');
            if(staticNotesDB[subjectName]) {
                renderTarget.innerHTML = staticNotesDB[subjectName];
            } else {
                renderTarget.innerHTML = `
                    <div class="chapter-title">Content Processing</div>
                    <div class="qa-card">
                        <p style="color: #64748b; font-style: italic;">We are currently uploading additional curriculum notes for Class ${SelectedClass} ${subjectName}. Please check back shortly!</p>
                    </div>`;
            }
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function closeNotesWindow() {
            document.getElementById('notesWindow').classList.remove('active-window');
